Código 9 – Criando galáxias de estrelas

Cria a função drawGalaxy(), que utiliza moveToRandomLocation() e drawStar() para desenhar galáxias com pequenas estrelas coloridas.

Python 3 · Turtle + Random
from turtle import *
from random import *

# Uma função para mover a tartaruga para uma posição aleatória
def moveToRandomLocation():
    penup()
    setpos(randint(-400, 400), randint(-400, 400))
    pendown()

# Uma função para desenhar uma estrela de um tamanho específico
def drawStar(starSize, starColour):
    color(starColour)
    pendown()
    begin_fill()
    for side in range(5):
        left(144)
        forward(starSize)
    end_fill()
    penup()

# Uma função para desenhar uma pequena galáxia de estrelas
def drawGalaxy(numberOfStars):
    starColours = ["#058396", "#0275A6", "#827E01"]
    moveToRandomLocation()

    # Desenha várias pequenas estrelas coloridas
    for star in range(numberOfStars):
        penup()
        left(randint(-180, 180))
        forward(randint(5, 20))
        pendown()

        # Desenha uma pequena estrela de cor aleatória
        drawStar(2, choice(starColours))

speed(11)

# Isso desenha um fundo azul escuro
bgcolor("MidnightBlue")

# Desenha 30 estrelas brancas
for star in range(30):
    moveToRandomLocation()
    drawStar(randint(5, 25), "White")

# Desenha 3 pequenas galáxias de 40 estrelas
for galaxy in range(3):
    drawGalaxy(40)

done()
ETAPA 4 · CONSTELAÇÕES

Código 10 – Desenhando constelações

Programa completo com estrelas aleatórias, três galáxias e duas constelações formadas por estrelas ligadas por linhas brancas.

Python 3 · Turtle + Random
from turtle import *
from random import *

# Uma função para mover a tartaruga para uma posição aleatória
def moveToRandomLocation():
    penup()
    setpos(randint(-400, 400), randint(-400, 400))
    pendown()

# Uma função para desenhar uma estrela de um tamanho específico
def drawStar(starSize, starColour):
    color(starColour)
    pendown()
    begin_fill()
    for side in range(5):
        left(144)
        forward(starSize)
    end_fill()
    penup()

# Uma função para desenhar uma pequena galáxia de estrelas
def drawGalaxy(numberOfStars):
    starColours = ["#058396", "#0275A6", "#827E01"]
    moveToRandomLocation()

    # Desenha várias pequenas estrelas coloridas
    for star in range(numberOfStars):
        penup()
        left(randint(-180, 180))
        forward(randint(5, 20))
        pendown()
        drawStar(2, choice(starColours))

# Uma função para desenhar uma constelação de estrelas
def drawConstellation(numberOfStars):
    moveToRandomLocation()

    # Primeiro desenhamos todas as estrelas, exceto a última,
    # conectadas por linhas
    for star in range(numberOfStars - 1):
        drawStar(randint(7, 15), "white")
        pendown()
        left(randint(-90, 90))
        forward(randint(30, 70))

    # Agora desenhamos a última estrela
    drawStar(randint(7, 15), "White")

speed(11)

# Isso desenha um fundo azul escuro
bgcolor("MidnightBlue")

# Desenha 30 estrelas brancas
for star in range(30):
    moveToRandomLocation()
    drawStar(randint(5, 25), "White")

# Desenha 3 pequenas galáxias de 40 estrelas
for galaxy in range(3):
    drawGalaxy(40)

# Desenha 2 constelações, cada uma com número aleatório de estrelas
for constellation in range(2):
    drawConstellation(randint(4, 7))

hideturtle()
done()

Código 11 – Desafio: criando um desenho próprio

Exemplo autoral para o desafio “Funções em toda parte!” das páginas 16 e 17. Desenha uma casa usando funções para quadrados, triângulos e retângulos.

Python 3 · Turtle
from turtle import *

# Função para desenhar um quadrado
def drawSquare(size, colorName):
    color(colorName)
    begin_fill()
    for side in range(4):
        forward(size)
        left(90)
    end_fill()

# Função para desenhar um triângulo
def drawTriangle(size, colorName):
    color(colorName)
    begin_fill()
    for side in range(3):
        forward(size)
        left(120)
    end_fill()

# Função para desenhar um retângulo
def drawRectangle(width, height, colorName):
    color(colorName)
    begin_fill()
    for side in range(2):
        forward(width)
        left(90)
        forward(height)
        left(90)
    end_fill()

# Configuração da tela
bgcolor("LightSkyBlue")
speed(5)
pensize(2)

# Desenha o sol
penup()
goto(250, 180)
pendown()
color("Yellow")
begin_fill()
circle(40)
end_fill()

# Desenha o corpo da casa
penup()
goto(-100, -100)
setheading(0)
pendown()
drawSquare(200, "Wheat")

# Desenha o telhado
penup()
goto(-100, 100)
setheading(0)
pendown()
drawTriangle(200, "Red")

# Desenha a porta
penup()
goto(-25, -100)
setheading(0)
pendown()
drawRectangle(50, 90, "SaddleBrown")

# Desenha a janela esquerda
penup()
goto(-80, 0)
setheading(0)
pendown()
drawSquare(45, "DeepSkyBlue")

# Desenha a janela direita
penup()
goto(35, 0)
setheading(0)
pendown()
drawSquare(45, "DeepSkyBlue")

# Finalização
hideturtle()
done()
